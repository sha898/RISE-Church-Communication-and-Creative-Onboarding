const BRANCH_COLORS = {
  Leadership: { bar: '#20242C', badgeBg: 'bg-slate-900 text-white dark:bg-slate-700' },
  Communications: { bar: '#2D6A4F', badgeBg: 'bg-emerald-800 text-white dark:bg-emerald-900/80' },
  Creative: { bar: '#9A4123', badgeBg: 'bg-orange-800 text-white dark:bg-orange-900/80' },
  Capture: { bar: '#6B46C1', badgeBg: 'bg-purple-800 text-white dark:bg-purple-900/80' },
  Online: { bar: '#1D4ED8', badgeBg: 'bg-blue-800 text-white dark:bg-blue-900/80' }
};

let currentRoleSlug = '';
let currentRoleObj = null;
let currentFilter = 'all';
let collapsedPhases = {};
let overviewBranchFilter = 'all';

function getRoleStorageKey(slug) {
  return 'rise_onboard_v2_' + slug;
}

function loadRoleState(roleObj) {
  const defaultState = { checked: {}, candidateName: '', startDate: '', customTasks: [], notes: {} };
  try {
    const raw = localStorage.getItem(getRoleStorageKey(roleObj.slug));
    if (raw) return { ...defaultState, ...JSON.parse(raw) };
  } catch (e) {
    console.error('Error loading state:', e);
  }
  return defaultState;
}

function saveRoleState(slug, state) {
  try {
    localStorage.setItem(getRoleStorageKey(slug), JSON.stringify(state));
  } catch (e) {
    console.error('Error saving state:', e);
  }
}

function initApp() {
  populateRoleDropdown();
  setupTheme();
  
  let savedRole = null;
  try { savedRole = localStorage.getItem('rise_onboard_lastRole'); } catch(e){}
  const startSlug = (savedRole && ROLES.some(r => r.slug === savedRole)) ? savedRole : ROLES[0].slug;
  
  document.getElementById('roleSelect').value = startSlug;
  handleRoleChange(startSlug);
}

function populateRoleDropdown() {
  const sel = document.getElementById('roleSelect');
  sel.innerHTML = '';

  const branches = [...new Set(ROLES.map(r => r.branch))];
  branches.forEach(b => {
    const og = document.createElement('optgroup');
    og.label = b + ' Branch';
    ROLES.filter(r => r.branch === b).forEach(r => {
      const opt = document.createElement('option');
      opt.value = r.slug;
      opt.textContent = r.title;
      og.appendChild(opt);
    });
    sel.appendChild(og);
  });
}

function handleRoleChange(slug) {
  currentRoleSlug = slug;
  try { localStorage.setItem('rise_onboard_lastRole', slug); } catch(e){}
  currentRoleObj = ROLES.find(r => r.slug === slug);

  const branchData = BRANCH_COLORS[currentRoleObj.branch] || BRANCH_COLORS.Leadership;
  document.getElementById('roleAccentBar').style.backgroundColor = branchData.bar;
  
  const badge = document.getElementById('roleBranchBadge');
  badge.textContent = currentRoleObj.branch + ' Branch';
  badge.className = `text-xs font-bold px-2.5 py-0.5 rounded-full ${branchData.badgeBg}`;

  document.getElementById('tierLabel').textContent = currentRoleObj.tier;
  document.getElementById('reportsTo').textContent = currentRoleObj.reportsTo;

  const state = loadRoleState(currentRoleObj);
  document.getElementById('assigneeInput').value = state.candidateName || '';
  document.getElementById('startDateInput').value = state.startDate || '';

  renderPhases();
  updateProgressStats();
}

function saveCandidateInfo() {
  if (!currentRoleObj) return;
  const state = loadRoleState(currentRoleObj);
  state.candidateName = document.getElementById('assigneeInput').value;
  state.startDate = document.getElementById('startDateInput').value;
  saveRoleState(currentRoleObj.slug, state);
}

function renderPhases() {
  const container = document.getElementById('phasesContainer');
  container.innerHTML = '';

  if (!currentRoleObj) return;
  const state = loadRoleState(currentRoleObj);
  const searchKey = (document.getElementById('taskSearchInput').value || '').toLowerCase().trim();

  currentRoleObj.phases.forEach((phaseTuple, phaseIdx) => {
    const phaseTitle = phaseTuple[0];
    const defaultItems = phaseTuple[1];

    const customItemsForPhase = (state.customTasks || []).filter(t => t.phaseIdx === phaseIdx);

    let phaseTotal = defaultItems.length + customItemsForPhase.length;
    let phaseDone = 0;

    defaultItems.forEach((_, itemIdx) => {
      if (state.checked[`d_${phaseIdx}_${itemIdx}`]) phaseDone++;
    });

    customItemsForPhase.forEach((_, cIdx) => {
      if (state.checked[`c_${phaseIdx}_${cIdx}`]) phaseDone++;
    });

    const phaseCard = document.createElement('div');
    phaseCard.className = 'bg-white dark:bg-paper-cardDark border border-slate-200 dark:border-slate-800 rounded-2xl overflow-hidden shadow-xs transition-all';

    const isCollapsed = collapsedPhases[phaseIdx];
    const header = document.createElement('div');
    header.className = 'px-5 py-4 bg-slate-50/80 dark:bg-slate-800/40 border-b border-slate-200/60 dark:border-slate-800 flex items-center justify-between cursor-pointer select-none';
    header.onclick = (e) => {
      if (e.target.closest('button')) return;
      collapsedPhases[phaseIdx] = !collapsedPhases[phaseIdx];
      renderPhases();
    };

    const phasePct = phaseTotal ? Math.round((phaseDone / phaseTotal) * 100) : 0;

    header.innerHTML = `
      <div class="flex items-center gap-3">
        <button class="w-6 h-6 rounded-md bg-slate-200 dark:bg-slate-700 flex items-center justify-center text-xs text-slate-600 dark:text-slate-300">
          <i class="fa-solid fa-chevron-${isCollapsed ? 'right' : 'down'}"></i>
        </button>
        <div>
          <h3 class="font-serif font-bold text-base text-slate-900 dark:text-white flex items-center gap-2">
            ${phaseTitle}
            ${phasePct === 100 ? '<i class="fa-solid fa-circle-check text-emerald-500 text-sm"></i>' : ''}
          </h3>
          <p class="text-xs text-slate-500 dark:text-slate-400 font-medium">${phaseDone} of ${phaseTotal} completed (${phasePct}%)</p>
        </div>
      </div>

      <div class="flex items-center gap-2">
        <button onclick="toggleAllPhaseItems(${phaseIdx}, true)" class="px-2.5 py-1 text-xs font-semibold rounded-lg bg-emerald-50 dark:bg-emerald-950/40 text-emerald-700 dark:text-emerald-400 hover:bg-emerald-100 transition">
          Mark All Done
        </button>
      </div>
    `;

    phaseCard.appendChild(header);

    if (!isCollapsed) {
      const listEl = document.createElement('div');
      listEl.className = 'p-3 sm:p-4 space-y-2';

      let renderedCount = 0;

      defaultItems.forEach((itemText, itemIdx) => {
        const itemKey = `d_${phaseIdx}_${itemIdx}`;
        const isChecked = !!state.checked[itemKey];

        if (currentFilter === 'completed' && !isChecked) return;
        if (currentFilter === 'pending' && isChecked) return;
        if (searchKey && !itemText.toLowerCase().includes(searchKey)) return;

        renderedCount++;
        listEl.appendChild(createTaskItemRow(itemText, itemKey, isChecked, false, phaseIdx, itemIdx));
      });

      customItemsForPhase.forEach((cItem, cIdx) => {
        const itemKey = `c_${phaseIdx}_${cIdx}`;
        const isChecked = !!state.checked[itemKey];

        if (currentFilter === 'completed' && !isChecked) return;
        if (currentFilter === 'pending' && isChecked) return;
        if (searchKey && !cItem.text.toLowerCase().includes(searchKey)) return;

        renderedCount++;
        listEl.appendChild(createTaskItemRow(cItem.text, itemKey, isChecked, true, phaseIdx, cIdx));
      });

      if (renderedCount === 0) {
        listEl.innerHTML = `<div class="text-center py-6 text-xs text-slate-400 italic">No tasks matching current filter or search.</div>`;
      }

      phaseCard.appendChild(listEl);
    }

    container.appendChild(phaseCard);
  });
}

function createTaskItemRow(itemText, itemKey, isChecked, isCustom, phaseIdx, itemIdx) {
  const row = document.createElement('div');
  row.className = `group p-3 rounded-xl border transition-all ${isChecked ? 'bg-slate-50/50 dark:bg-slate-800/20 border-slate-200/40 dark:border-slate-800' : 'bg-white dark:bg-slate-900 border-slate-200 dark:border-slate-800 hover:border-slate-300 dark:hover:border-slate-700'}`;

  const state = loadRoleState(currentRoleObj);
  const noteText = state.notes[itemKey] || '';

  row.innerHTML = `
    <div class="flex items-start gap-3">
      <input type="checkbox" ${isChecked ? 'checked' : ''} onchange="toggleTaskChecked('${itemKey}', this.checked)" class="mt-1 w-5 h-5 rounded border-slate-300 dark:border-slate-700 accent-slate-900 dark:accent-amber-500 cursor-pointer">
      
      <div class="flex-1 min-w-0">
        <div class="flex items-start justify-between gap-2">
          <p class="text-sm leading-relaxed ${isChecked ? 'line-through text-slate-400 dark:text-slate-500' : 'text-slate-800 dark:text-slate-200 font-medium'}">
            ${itemText}
            ${isCustom ? '<span class="ml-2 text-[10px] font-bold uppercase tracking-wider px-2 py-0.5 rounded bg-amber-100 text-amber-800 dark:bg-amber-950 dark:text-amber-300">Custom</span>' : ''}
          </p>

          <div class="flex items-center gap-1 opacity-80 sm:opacity-0 group-hover:opacity-100 transition">
            <button onclick="toggleNoteBox('${itemKey}')" title="${noteText ? 'Edit Note' : 'Add Note'}" class="p-1.5 rounded-lg text-slate-400 hover:text-slate-700 dark:hover:text-slate-200 hover:bg-slate-100 dark:hover:bg-slate-800 text-xs">
              <i class="fa-solid fa-sticky-note ${noteText ? 'text-amber-500' : ''}"></i>
            </button>
            ${isCustom ? `
              <button onclick="deleteCustomTask(${phaseIdx}, ${itemIdx})" title="Delete Custom Task" class="p-1.5 rounded-lg text-slate-400 hover:text-rose-500 hover:bg-rose-50 dark:hover:bg-rose-950/40 text-xs">
                <i class="fa-solid fa-trash-can"></i>
              </button>
            ` : ''}
          </div>
        </div>

        ${noteText ? `<div class="mt-2 text-xs text-amber-800 dark:text-amber-200 bg-amber-50 dark:bg-amber-950/30 p-2.5 rounded-lg border border-amber-200/50 dark:border-amber-900/40 flex items-start gap-2">
          <i class="fa-solid fa-comment text-amber-500 mt-0.5"></i>
          <span class="whitespace-pre-wrap">${escapeHtml(noteText)}</span>
        </div>` : ''}

        <div id="noteBox_${itemKey}" class="hidden mt-3 space-y-2">
          <textarea id="noteInput_${itemKey}" rows="2" placeholder="Add private onboarding notes or links..." class="w-full text-xs p-2.5 bg-slate-50 dark:bg-slate-800 border border-slate-200 dark:border-slate-700 rounded-lg text-slate-800 dark:text-slate-200 focus:outline-none focus:ring-1 focus:ring-amber-500">${escapeHtml(noteText)}</textarea>
          <div class="flex items-center justify-end gap-2">
            <button onclick="toggleNoteBox('${itemKey}')" class="px-2.5 py-1 text-xs text-slate-500">Cancel</button>
            <button onclick="saveNote('${itemKey}')" class="px-3 py-1 bg-slate-900 dark:bg-amber-500 text-white dark:text-slate-950 rounded-md text-xs font-bold">Save Note</button>
          </div>
        </div>

      </div>
    </div>
  `;

  return row;
}

function escapeHtml(text) {
  return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
}

function toggleTaskChecked(itemKey, isChecked) {
  if (!currentRoleObj) return;
  const state = loadRoleState(currentRoleObj);
  state.checked[itemKey] = isChecked;
  saveRoleState(currentRoleObj.slug, state);

  updateProgressStats();
  renderPhases();

  const total = calculateRoleTotalTasks(currentRoleObj, state);
  const done = calculateRoleDoneTasks(currentRoleObj, state);
  if (done === total && total > 0 && isChecked) {
    triggerConfetti();
    showToast(`🎉 Onboarding complete for ${currentRoleObj.title}!`, 'success');
  }
}

function toggleAllPhaseItems(phaseIdx, markAll) {
  if (!currentRoleObj) return;
  const state = loadRoleState(currentRoleObj);
  const phaseTuple = currentRoleObj.phases[phaseIdx];
  const defaultItems = phaseTuple[1];
  const customItems = (state.customTasks || []).filter(t => t.phaseIdx === phaseIdx);

  defaultItems.forEach((_, itemIdx) => {
    state.checked[`d_${phaseIdx}_${itemIdx}`] = markAll;
  });
  customItems.forEach((_, cIdx) => {
    state.checked[`c_${phaseIdx}_${cIdx}`] = markAll;
  });

  saveRoleState(currentRoleObj.slug, state);
  updateProgressStats();
  renderPhases();
  showToast('Phase marked as complete', 'success');
}

function toggleNoteBox(itemKey) {
  const box = document.getElementById(`noteBox_${itemKey}`);
  if (box) box.classList.toggle('hidden');
}

function saveNote(itemKey) {
  if (!currentRoleObj) return;
  const state = loadRoleState(currentRoleObj);
  const input = document.getElementById(`noteInput_${itemKey}`);
  if (input) {
    state.notes[itemKey] = input.value.trim();
    saveRoleState(currentRoleObj.slug, state);
    renderPhases();
    showToast('Note saved successfully', 'success');
  }
}

function updateProgressStats() {
  if (!currentRoleObj) return;
  const state = loadRoleState(currentRoleObj);
  const total = calculateRoleTotalTasks(currentRoleObj, state);
  const done = calculateRoleDoneTasks(currentRoleObj, state);
  const pct = total ? Math.round((done / total) * 100) : 0;

  document.getElementById('progressFill').style.width = pct + '%';
  document.getElementById('progressCount').textContent = `${done} of ${total} (${pct}%)`;

  const badge = document.getElementById('completionBadge');
  if (pct === 100 && total > 0) {
    badge.textContent = 'Complete';
    badge.className = 'text-xs font-sans px-2.5 py-0.5 rounded-full bg-emerald-100 text-emerald-800 dark:bg-emerald-950 dark:text-emerald-300 font-bold';
  } else if (pct > 0) {
    badge.textContent = 'In Progress';
    badge.className = 'text-xs font-sans px-2.5 py-0.5 rounded-full bg-amber-100 text-amber-800 dark:bg-amber-950 dark:text-amber-300 font-bold';
  } else {
    badge.textContent = 'Not Started';
    badge.className = 'text-xs font-sans px-2.5 py-0.5 rounded-full bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-400 font-bold';
  }
}

function calculateRoleTotalTasks(roleObj, state) {
  let count = 0;
  roleObj.phases.forEach(p => count += p[1].length);
  count += (state.customTasks || []).length;
  return count;
}

function calculateRoleDoneTasks(roleObj, state) {
  let done = 0;
  roleObj.phases.forEach((p, pIdx) => {
    p[1].forEach((_, iIdx) => {
      if (state.checked[`d_${pIdx}_${iIdx}`]) done++;
    });
  });
  (state.customTasks || []).forEach((cTask, cIdx) => {
    if (state.checked[`c_${cTask.phaseIdx}_${cIdx}`]) done++;
  });
  return done;
}

function setFilter(filterType) {
  currentFilter = filterType;
  ['All', 'Pending', 'Completed'].forEach(f => {
    const btn = document.getElementById(`filter${f}Btn`);
    if (f.toLowerCase() === filterType) {
      btn.className = 'px-2.5 py-1 rounded-md font-medium transition bg-slate-900 dark:bg-amber-500 text-white dark:text-slate-950';
    } else {
      btn.className = 'px-2.5 py-1 rounded-md font-medium text-slate-600 dark:text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-800 transition';
    }
  });
  renderPhases();
}

  const sel = document.getElementById('modalPhaseSelect');
  sel.innerHTML = '';
  currentRoleObj.phases.forEach((p, idx) => {
    const opt = document.createElement('option');
    opt.value = idx;
    opt.textContent = p[0];
    sel.appendChild(opt);
  });

  document.getElementById('addTaskModal').classList.remove('hidden');
}

function closeAddTaskModal() {
  document.getElementById('addTaskModal').classList.add('hidden');
}

function submitCustomTask() {
  const phaseIdx = parseInt(document.getElementById('modalPhaseSelect').value);
  const text = document.getElementById('modalTaskInput').value.trim();

  if (!text) {
    showToast('Please enter a task description', 'error');
    return;
  }

  const state = loadRoleState(currentRoleObj);
  if (!state.customTasks) state.customTasks = [];
  state.customTasks.push({ phaseIdx, text });

  saveRoleState(currentRoleObj.slug, state);
  closeAddTaskModal();
  renderPhases();
  updateProgressStats();
  showToast('Custom task added', 'success');
}

function deleteCustomTask(phaseIdx, customIdx) {
  const state = loadRoleState(currentRoleObj);
  state.customTasks.splice(customIdx, 1);
  saveRoleState(currentRoleObj.slug, state);
  renderPhases();
  updateProgressStats();
  showToast('Custom task removed', 'success');
}

function confirmResetRole() {

    closeConfirmModal();

  };

}

}

  });
  renderOverviewDashboard();
}

function renderOverviewDashboard() {
  const grid = document.getElementById('overviewCardsGrid');
  grid.innerHTML = '';

  let activeCount = 0;
  let totalTasksDoneSum = 0;
  let totalTasksPossibleSum = 0;

  const filteredRoles = ROLES.filter(r => overviewBranchFilter === 'all' || r.branch === overviewBranchFilter);

  filteredRoles.forEach(roleObj => {
    const state = loadRoleState(roleObj);
    const total = calculateRoleTotalTasks(roleObj, state);
    const done = calculateRoleDoneTasks(roleObj, state);
    const pct = total ? Math.round((done / total) * 100) : 0;

    totalTasksDoneSum += done;
    totalTasksPossibleSum += total;

    if (done > 0 || state.candidateName) activeCount++;

    const branchData = BRANCH_COLORS[roleObj.branch] || BRANCH_COLORS.Leadership;

    const card = document.createElement('div');
    card.className = 'bg-white dark:bg-paper-cardDark border border-slate-200 dark:border-slate-800 rounded-2xl p-5 shadow-xs hover:border-slate-300 dark:hover:border-slate-700 transition flex flex-col justify-between';

    card.innerHTML = `
      <div class="space-y-3">
        <div class="flex items-center justify-between gap-2">
          <span class="text-[11px] font-bold px-2.5 py-0.5 rounded-full ${branchData.badgeBg}">${roleObj.branch}</span>
          <span class="text-xs font-bold ${pct === 100 ? 'text-emerald-500' : 'text-slate-500'}">${pct}%</span>
        </div>

        <div>
          <h3 class="font-serif font-bold text-base text-slate-900 dark:text-white leading-snug">${roleObj.title}</h3>
          <p class="text-xs text-slate-500 dark:text-slate-400 mt-0.5">
            ${state.candidateName ? `<i class="fa-solid fa-user text-amber-500 mr-1"></i><b>${escapeHtml(state.candidateName)}</b>` : 'Unassigned'}
          </p>
        </div>

        <div class="space-y-1">
          <div class="w-full h-2 bg-slate-100 dark:bg-slate-800 rounded-full overflow-hidden">
            <div class="h-full bg-slate-900 dark:bg-amber-500 transition-all duration-300" style="width: ${pct}%"></div>
          </div>
          <p class="text-[11px] text-slate-400">${done} of ${total} tasks finished</p>
        </div>
      </div>

      <div class="pt-4 mt-3 border-t border-slate-100 dark:border-slate-800/80 flex items-center justify-between">
        <span class="text-[11px] text-slate-400">Reports: ${roleObj.reportsTo}</span>
        <button onclick="jumpToRole('${roleObj.slug}')" class="px-3 py-1.5 bg-slate-100 dark:bg-slate-800 hover:bg-slate-200 dark:hover:bg-slate-700 rounded-lg text-xs font-bold text-slate-800 dark:text-slate-200 transition">
          Open <i class="fa-solid fa-arrow-right ml-1"></i>
        </button>
      </div>
    `;

    grid.appendChild(card);
  });

  document.getElementById('statTotalRoles').textContent = ROLES.length;
  document.getElementById('statActiveCount').textContent = activeCount;
  document.getElementById('statTotalTasksDone').textContent = totalTasksDoneSum;
  
  const overallPct = totalTasksPossibleSum ? Math.round((totalTasksDoneSum / totalTasksPossibleSum) * 100) : 0;
  document.getElementById('statOverallPct').textContent = overallPct + '%';
}

function jumpToRole(slug) {
  switchTab('checklist');
  document.getElementById('roleSelect').value = slug;
  handleRoleChange(slug);
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

function copyFormattedSummary() {
  if (!currentRoleObj) return;
  const state = loadRoleState(currentRoleObj);
  const total = calculateRoleTotalTasks(currentRoleObj, state);
  const done = calculateRoleDoneTasks(currentRoleObj, state);
  const pct = total ? Math.round((done / total) * 100) : 0;

  let text = `📋 *Rise Church Onboarding Status*\n`;
  text += `*Role:* ${currentRoleObj.title}\n`;
  if (state.candidateName) text += `*Candidate:* ${state.candidateName}\n`;
  if (state.startDate) text += `*Start Date:* ${state.startDate}\n`;
  text += `*Progress:* ${done}/${total} tasks (${pct}%)\n\n`;

  currentRoleObj.phases.forEach((phaseTuple, pIdx) => {
    text += `*${phaseTuple[0]}:*\n`;
    phaseTuple[1].forEach((item, iIdx) => {
      const isChecked = state.checked[`d_${pIdx}_${iIdx}`];
      text += ` ${isChecked ? '✅' : '⏹️'} ${item}\n`;
    });
    text += `\n`;
  });

  copyToClipboard(text);
  showToast('Formatted summary copied to clipboard!', 'success');
}

function copyToClipboard(text) {
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(text);
  } else {
    const textarea = document.createElement('textarea');
    textarea.value = text;
    document.body.appendChild(textarea);
    textarea.select();
    document.execCommand('copy');
    document.body.removeChild(textarea);
  }
}

function exportAllData() {
  const exportObject = {};
  ROLES.forEach(r => {
    exportObject[r.slug] = loadRoleState(r);
  });

  const blob = new Blob([JSON.stringify(exportObject, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `Rise_Church_Onboarding_Backup_${new Date().toISOString().slice(0,10)}.json`;
  a.click();
  showToast('Backup file downloaded', 'success');
}

function importAllData(event) {
  const file = event.target.files[0];
  if (!file) return;

  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const imported = JSON.parse(e.target.result);
      Object.keys(imported).forEach(slug => {
        saveRoleState(slug, imported[slug]);
      });
      showToast('Data restored successfully!', 'success');
      if (currentRoleSlug) handleRoleChange(currentRoleSlug);
      renderOverviewDashboard();
    } catch(err) {
      showToast('Invalid backup file', 'error');
    }
  };
  reader.readAsText(file);
}

function setupTheme() {
  if (localStorage.getItem('theme') === 'dark' || (!('theme' in localStorage) && window.matchMedia('(prefers-color-scheme: dark)').matches)) {
    document.documentElement.classList.add('dark');
    document.getElementById('themeIcon').className = 'fa-solid fa-sun text-amber-400';
  } else {
    document.documentElement.classList.remove('dark');
    document.getElementById('themeIcon').className = 'fa-solid fa-moon text-slate-600';
  }
}

function toggleDarkMode() {
  if (document.documentElement.classList.contains('dark')) {
    document.documentElement.classList.remove('dark');
    localStorage.setItem('theme', 'light');
    document.getElementById('themeIcon').className = 'fa-solid fa-moon text-slate-600';
  } else {
    document.documentElement.classList.add('dark');
    localStorage.setItem('theme', 'dark');
    document.getElementById('themeIcon').className = 'fa-solid fa-sun text-amber-400';
  }
}

}

function showToast(msg, type = 'success') {
  const toast = document.getElementById('toast');
  const toastMsg = document.getElementById('toastMsg');
  const toastIcon = document.getElementById('toastIcon');

  toastMsg.textContent = msg;
  if (type === 'success') {
    toastIcon.className = 'fa-solid fa-circle-check text-emerald-400 text-base';
  } else {
    toastIcon.className = 'fa-solid fa-circle-exclamation text-rose-400 text-base';
  }

  toast.classList.remove('translate-y-20', 'opacity-0');
  toast.classList.add('translate-y-0', 'opacity-100');

  setTimeout(() => {
    toast.classList.add('translate-y-20', 'opacity-0');
    toast.classList.remove('translate-y-0', 'opacity-100');
  }, 2500);
}
